// CLASSES

class Host {
  construct new(address, ports) {
    _address = address
    _ports = ports
  }

  address {
    return _address
  }

  ports {
    return _ports
  }

  describe() {
    System.print("Host: %(_address)")
  }
}

var host = Host.new("1.1.1.1", [80, 443])

host.describe()

System.print(host.address)
